[unix ODBC]
brew install unixODBC unixODBC-devel

[MSSQL Driver for ODBC]
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/master/install.sh)"
# brew tap microsoft/mssql-release https://github.com/Microsoft/homebrew-mssql-release
# brew install msodbcsql18 mssql-tools18

# vi /opt/homebrew/etc/odbcinst.ini
<<
Encrypt=Optional
TrustServerCertificate=Yes

# odbcinst -j
# odbcinst -q -d

--Declare in Makefile:
$(INC_ODBC) = /opt/Homebrew/include
$(LIB_ODBC) = /opt/Homebrew/lib

[Oracle Client and Library]
--Download from Oracle site: 
https://www.oracle.com/cn/database/technologies/instant-client/macos-downloads.html Basic Light (dmg)  SDK (dmg) 
Make directory and Copy the Oracle Package to /opt/oracle/instantclient_23i

# sudo vi ~/.bash_profile
<<
export ORACLE_HOME=/opt/oracle/instantclient_23i
export TNS_ADMIN=$ORACLE_HOME/network/admin
export NLS_LANG=SIMPLIFIED_CHINESE_CHINA.ALUTF8
export LD_LIBRARY_PATH=$ORACLE_HOME
export SQLPATH=$ORACLE_HOME
export PATH=$ORACLE_HOME:$PATH
unset no_proxy
unset NLS_LANG

# source .bash_profile

--Declare in Makefile:
$(INC_OCI) = /opt/oracle/instantclient_23i/sdk/include
$(LIB_OCI) = /opt/oracle/instantclient_23i
