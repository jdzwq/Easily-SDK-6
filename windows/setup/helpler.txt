---------------------------------------------------------------------------------
        INSTALL DATABASE CONNECT TOOLS AND LIBRARY FOR WINDOWS11 ARM64
---------------------------------------------------------------------------------

[MSVC ODBC]
Install the Visual Studio Command Prompt Tools with Windows SDK

[MSSQL Driver For ODBC]
Install the SQL Server with ODBC Drivers

netsh advfirewall firewall add rule name="SQL Server Port" dir=in action=allow protocol=TCP localport=1433

[Mysql Client and Library]
--Download from Mysql site:
https://downloads.mysql.com/archives/c-c/
Install the MySql Connect Library and Client Tools to E:/mysql

--Execute in PowerShell:
[Environment]::SetEnvironmentVariable("PATH", $env:PATH + ";E:\mysql\bin",[System.EnvironmentVariableTarget]::Machine) 

--Declare in Makefile:
$(INC_MYSQL) = E:\mysql\include
$(LIB_MYSQL) = E:\mysql\lib\vc14

[Postgres Client and Library]
--Download from Postgresql Site:
https://www.enterprisedb.com/download-postgresql-binaries >> Win x86-64
Copy the Pgsql Package to E:\pgsql

--Execute in PowerShell:
[Environment]::SetEnvironmentVariable("PATH", $env:PATH + ";E:\pgsql\bin",[System.EnvironmentVariableTarget]::Machine) 

--Declare in Makefile
$(INC_POSTGRE) = E:\pgsql\include
$(LIB_POSTGRE) = E:\pgsql\lib


[Oracle Client and Library]
--Download from Oracle site: 
https://www.oracle.com/cn/database/technologies/instant-client/winx64-64-downloads.html >> instantclient-basic-windows.x64-19 instantclient-sqlplus-windows.x64-19
Make directory and Copy the Oracle Package to E:\oracle

--Execurte in PowerShell:
[Environment]::SetEnvironmentVariable("ORACLE_HOME", "E:\oracle\instantclient_19c", [System.EnvironmentVariableTarget]::Machine)
[Environment]::SetEnvironmentVariable("NLS_LANG", "SIMPLIFIED_CHINESE_CHINA.ALUTF8", [System.EnvironmentVariableTarget]::Machine)
[Environment]::SetEnvironmentVariable("PATH", $env:PATH + ";E:\oracle\instantclient_19c",[System.EnvironmentVariableTarget]::Machine) 

--Declare in Makefile:
$(INC_ORACLE) = E:\oracle\instantclient_19c\sdk\include
$(LIB_ORACLE) = E:\oracle\instantclient_19c\sdk\lib\msvc


[Sqlite Client and Library]
--Download from Sqlite site:
https://www.sqlite.org/download.html >> sqlite-src sqlite-amalgamation sqlite-tools-win-x64 sqlite-dll-win-x64

--Extract sqlite-src to E:/sqlite3
--Build Library for Windows x64
E:/sqlite3> nmake /f Makefile.msc
E:/sqlite3> copy sqlite3.lib E:/sqlite/lib

--Execute in PowerShell:
[Environment]::SetEnvironmentVariable("PATH", $env:PATH + ";E:\sqlite\bin",[System.EnvironmentVariableTarget]::Machine) 

--Declare in Makefile:
$(INC_SQLITE) = E:\sqlite\include
$(LIB_SQLITE) = E:\sqlite\lib
